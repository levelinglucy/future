// ===== Seeded RNG =====
function xmur3(str){
  let h=1779033703^str.length;
  for(let i=0;i<str.length;i++){ h=Math.imul(h^str.charCodeAt(i),3432918353); h=(h<<13)|(h>>>19); }
  return function(){ h=Math.imul(h^(h>>>16),2246822507); h=Math.imul(h^(h>>>13),3266489909); return (h^=(h>>>16))>>>0; }
}
function mulberry32(a){
  return function(){
    let t=a+=0x6D2B79F5;
    t=Math.imul(t^(t>>>15),t|1);
    t^=t+Math.imul(t^(t>>>7),t|61);
    return ((t^(t>>>14))>>>0)/4294967296;
  }
}
let seedStr="ELYTHIA-"+Math.floor(Math.random()*1e9);
function rngFor(key){ const s=xmur3(seedStr+"|"+key)(); return mulberry32(s); }

let sigils={}; // lineageId -> symbol
function makeSigil(lineage){
  if(sigils[lineage]) return sigils[lineage];
  const r=rngFor("sigil|"+lineage);
  let s="";
  s += glyphAlphabet[(r()*glyphAlphabet.length)|0];
  s += glyphAlphabet[(r()*glyphAlphabet.length)|0];
  sigils[lineage]=s;
  return s;
}

// ===== Relics (inventory, active, refusal, fusion, evolution) =====
function normAura(a){
  const m=a.A+a.B+a.C+1e-9;
  return {A:a.A/m,B:a.B/m,C:a.C/m};
}
function auraDot(a,b){ return a.A*b.A + a.B*b.B + a.C*b.C; }
function blendAura(a,b){ return normAura({A:(a.A+b.A)/2, B:(a.B+b.B)/2, C:(a.C+b.C)/2}); }

function newRelic(name, aura, mods){
  const r=rngFor("relic|"+name+"|"+Math.random());
  const id="R"+Math.floor(r()*1e9);
  const sig= glyphAlphabet[(r()*glyphAlphabet.length)|0] + glyphAlphabet[(r()*glyphAlphabet.length)|0];
  return {
    id, name, sigil:sig,
    aura:normAura(aura),
    mods:{...mods},
    fusedFrom:null,
    rejectedBy:{},
    lvl:1,
    charge:0,
    quirks:[]
  };
}

let relics=[];

function cloneTemplate(t){
  return {
    id:t.id+"_"+Math.floor(Math.random()*1e6),
    name:t.name,
    sigil:t.sigil,
    aura:{...t.aura},
    mods:{...t.mods},
    fusedFrom:null,
    rejectedBy:{},

    charge:0,
    quirks:[]
  };
}

function grantRandomRelic(){
  const pick = relicTemplates[(Math.random()*relicTemplates.length)|0];
  const r = cloneTemplate(pick);
  relics.push(r);
  if(!activeRelicId) activeRelicId=r.id;
  toast("Relic gained: "+r.name+" "+r.sigil);
}

function getActiveRelic(){
  if(!activeRelicId) return null;
  return relics.find(r=>r.id===activeRelicId) || null;
}

// Realm refusal: per-relic per-realm persistent
function isRelicRejectedInRealm(relic, dim){
  if(!relic) return false;
  return !!relic.rejectedBy[dim.id];
}
function maybeRefuseRelic(relic, dim){
  if(!relic) return false;
  if(relic.rejectedBy[dim.id]) return true;

  // mismatch + gravity + age -> refusal chance
  const sim = auraDot(relic.aura, normAura(dim.biome)); // 0..1
  const grav = computeGravity(dim);
  const age = realmAge(dim);
  const stubborn = clamp(age/1400, 0, 0.45);
  const chance = clamp((0.56 - sim)*0.72 + (grav*0.18) + stubborn, 0, 0.88);

  if(Math.random() < chance){
    relic.rejectedBy[dim.id] = true;
    mythPush(dim, `The ${relic.name} refuses this realm.`);
    return true;
  }
  return false;
}

function fuseRelics(a, b){
  const fused = cloneTemplate(a);
  fused.name = a.name.split(" ")[0] + " " + b.name.split(" ")[1];
  fused.aura = blendAura(a.aura, b.aura);
  fused.mods = {...a.mods, ...b.mods}; // simplistic merge
  fused.lvl = Math.max(a.lvl, b.lvl) + 1;
  fused.fusedFrom = [a.id, b.id];
  fused.sigil = makeSigil(fused.id);

  const idxA = relics.findIndex(r=>r.id===a.id);
  const idxB = relics.findIndex(r=>r.id===b.id);
  relics.splice(Math.max(idxA,idxB),1);
  relics.splice(Math.min(idxA,idxB),1);
  relics.push(fused);
  activeRelicId = fused.id;

  toast("Fusion: " + fused.name + " " + fused.sigil + " L" + fused.lvl);
  return fused;
}

let dimensions = {};

function makeTile(t){
  return {t, h: t===VOID ? 0 : 0.5};

function createDimension(id, parentId, portalPos, isEcho=false){
  const r = rngFor("dim|"+id);
  const biome = {
    A: r()*0.4 + 0.3,
    B: r()*0.4 + 0.3,
    C: r()*0.4 + 0.3
  };
  const sum = biome.A + biome.B + biome.C;
  biome.A /= sum; biome.B /= sum; biome.C /= sum;

  const tiles = Array(SIZE).fill().map(()=>Array(SIZE).fill().map(()=>makeTile(VOID)));

  // initial mycelium bloom from center or portal
  const cx = isEcho ? SIZE/2 : (portalPos ? portalPos.x : SIZE/2|0);
  const cy = isEcho ? SIZE/2 : (portalPos ? portalPos.y : SIZE/2|0);
  tiles[cy][cx].t = MYC;
  tiles[cy][cx].h = 1;

  dimensions[id] = {
    id,
    name: dimName(id),
    temperament: r() > 0.5 ? "expansive" : "contemplative",
    biome,
    tiles,
    entities: [],
    portals: [],
    portalBack: parentId ? {x:cx, y:cy, target:parentId} : null,
    myths: null,
    history: {visits:0, linger:0, touches:0, templeRites:0, fastTravels:0, gravity:0, lastVisitTick:0},
    discovered: true
  };

  // place a temple somewhere
  const tr = rngFor("temple|"+id);
  const tx = clamp(cx + ((tr()*20)|0)-10, 5, SIZE-6);
  const ty = clamp(cy + ((tr()*20)|0)-10, 5, SIZE-6);
  tiles[ty][tx].t = TEMPLE;
  dimensions[id].portals.push({x:tx, y:ty, target:null});

  if(parentId){
    const parent = dimensions[parentId];
    parent.portals.find(p=>p.target===null).target = id;
  }

  return dimensions[id];
}

function findPortalAt(dim, x, y){
  return dim.portals.find(p=>p.x===x && p.y===y);
}

function realmAge(d){
  return tick - d.history.lastVisitTick;
}

function computeGravity(d){
  return clamp(d.history.visits*0.08 + d.history.linger*0.004 + d.history.touches*0.12, 0, 2.8);
}

// ===== Evolution =====
function evolveDimension(d, isActive){
  const rate = isActive ? 1.2 : 0.4;
  for(let y=0;y<SIZE;y++){
    for(let x=0;x<SIZE;x++){
      const t = d.tiles[y][x];
      if(t.t === MYC && Math.random() < 0.008*rate){
        const [dx,dy] = dirs[(Math.random()*4)|0];
        const nx=x+dx, ny=y+dy;
        if(inB(nx,ny) && d.tiles[ny][nx].t===VOID){
          d.tiles[ny][nx].t = MYC;
          d.tiles[ny][nx].h = 0.3 + Math.random()*0.4;
        }
      }
    }
  }
}

function relicChargeTick(d){
  const relic = getActiveRelic();
  if(!relic) return;
  if(Math.random() < 0.02) relic.charge = Math.min(relic.charge+1, 100);
}

// ===== Drawing =====
function drawWorld(){
  g.clearRect(0,0,innerWidth,innerHeight);
  const d = dimensions[player.dimId] || dimensions["root"];
  const offsX = innerWidth/2 - player.x*CELL;
  const offsY = innerHeight/2 - player.y*CELL;

  // background gradient
  const grad = g.createRadialGradient(innerWidth/2, innerHeight/2, 0, innerWidth/2, innerHeight/2, Math.max(innerWidth,innerHeight));
  grad.addColorStop(0, "#0a1420");
  grad.addColorStop(1, "#000306");
  g.fillStyle = grad;
  g.fillRect(0,0,innerWidth,innerHeight);

  // tiles
  for(let y=0;y<SIZE;y++){
    for(let x=0;x<SIZE;x++){
      const t = d.tiles[y][x];
      const px = offsX + x*CELL + CELL/2;
      const py = offsY + y*CELL + CELL/2;
      if(t.t === VOID) continue;

      let col = "#111";
      if(t.t === MYC) col = "#1a3a2a";
      if(t.t === STRUCT) col = "#2a2a3a";
      if(t.t === TEMPLE) col = "#3a2a4a";
      if(t.t === PATH) col = "#223344";

      g.fillStyle = col;
      g.fillRect(offsX+x*CELL, offsY+y*CELL, CELL, CELL);

      // height glow
      g.globalAlpha = t.h * 0.6;
      g.fillStyle = "#ffffff";
      g.fillRect(offsX+x*CELL, offsY+y*CELL, CELL, CELL);
      g.globalAlpha = 1;
    }
  }

  // portals / seams
  d.portals.forEach(p=>{
    const px = offsX + p.x*CELL + CELL/2;
    const py = offsY + p.y*CELL + CELL/2;
    g.beginPath();
    g.arc(px,py,CELL*0.8,0,Math.PI*2);
    g.strokeStyle = "rgba(100,180,255,0.4)";
    g.lineWidth = 2;
    g.stroke();
  });

  // player glow pulse
  const mods = {relic: getActiveRelic(), refused: false};
  if(mods.relic && isRelicRejectedInRealm(mods.relic, d)) mods.refused = true;

  g.beginPath();
  g.arc(innerWidth/2, innerHeight/2, CELL*0.34, 0, Math.PI*2);
  g.fillStyle=`rgba(0,255,255,${0.30 + ((Math.sin(pulse)+1)/2)*0.25})`;
  g.fill();
  g.strokeStyle="rgba(255,255,255,0.25)";
  g.stroke();

  // active relic ring
  if(mods.relic){
    g.beginPath();
    g.arc(innerWidth/2, innerHeight/2, CELL*0.70, 0, Math.PI*2);
    g.strokeStyle=`rgba(255,255,255,${mods.refused?0.10:0.22})`;
    g.stroke();
  }

  // echo overlay
  if(d.id==="echo"){
    g.fillStyle="rgba(180,200,255,0.05)";
    g.fillRect(0,0,innerWidth,innerHeight);
  }

  d.history.gravity = computeGravity(d);
  const dom=domBiome(d.biome);
  const ageLabel = n=> n<60?"newborn":n<300?"young":n<900?"mature":"ancient";
  const age = realmAge(d);
  const relicLabel = mods.relic ? `${mods.relic.sigil} L${mods.relic.lvl} ${mods.relic.name}${mods.refused?" (refused)":" "}` : "none";
  statusEl.textContent = `${d.name} · ${dom} · age ${ageLabel(age)} · gravity ${d.history.gravity.toFixed(2)} · relic: ${relicLabel}`;
}

// ===== Star Map =====
function drawStarMap(){
  g.fillStyle = "rgba(0,0,0,0.75)";
  g.fillRect(0,0,innerWidth,innerHeight);

  const cx = innerWidth/2, cy = innerHeight/2;
  const maxRad = Math.min(innerWidth,innerHeight)*0.38;

  Object.keys(dimensions).forEach(id=>{
    if(id==="echo" && !echoUnlocked) return;
    const d = dimensions[id];
    const pos = starPos(id);
    const x = cx + pos.x * maxRad;
    const y = cy + pos.y * maxRad;
    d._mapX = x; d._mapY = y; d._mapR = 18;

    const pulseSize = 12 + Math.sin(pulse + id.charCodeAt(0))*4;
    g.beginPath();
    g.arc(x,y,pulseSize,0,Math.PI*2);
    g.fillStyle = id===player.dimId ? "rgba(0,255,255,0.8)" : "rgba(255,255,255,0.4)";
    g.fill();

    g.fillStyle = "white";
    g.font = "11px monospace";
    g.textAlign = "center";
    g.fillText(d.name, x, y+28);
  });
}

// ===== Interaction =====
function interact(){
  const d = dimensions[player.dimId];
  const tile = d.tiles[player.y][player.x];

  if(tile.t === TEMPLE){
    mythPush(d, "You perform a quiet rite in the temple light.");
    d.history.templeRites++;
    toast("Temple rite performed ✨");
    if(Math.random()<0.25) grantRandomRelic();
  }

  const portal = findPortalAt(d, player.x, player.y);
  if(portal && portal.target){
    player.dimId = portal.target;
    const targetDim = dimensions[portal.target];
    targetDim.history.visits++;
    targetDim.history.lastVisitTick = tick;
    mythPush(targetDim, "You step through the seam into a new rhythm.");
    toast("Entered: " + targetDim.name);
  }
}

function move(dx,dy){
  const nx = player.x + dx, ny = player.y + dy;
  if(inB(nx,ny)){
    player.x = nx; player.y = ny;
    const d = dimensions[player.dimId];
    d.history.touches++;
    if(d.tiles[ny][nx].t === MYC){
      travelSig.steps++;

function attemptFuse(){
  if(relics.length<2){
    toast("Need 2+ relics to fuse 🧩");
    return;
  }
  const active=getActiveRelic();
  if(!active){
    activeRelicId=relics[0].id;
    toast("Pick an active relic first 🗝️");
    return;
  }
  const idx=relics.findIndex(r=>r.id===activeRelicId);
  const other=relics[(idx+1)%relics.length];
  if(!other || other.id===active.id){
    toast("No valid fuse target");
    return;
  }
  const fused=fuseRelics(active, other);
  const d=dimensions[player.dimId];
  mythPush(d, `Two laws fuse: ${fused.name} is born.`);
  addJournal("Fusion rite complete. A new law forms. 🧩✨");
}

function rleEncodeTiles(tiles){
  const rows=[];
  for(let y=0;y<SIZE;y++){
    let row="";
    let last=tiles[y][0].t;
    let count=1;
    for(let x=1;x<SIZE;x++){
      const t=tiles[y][x].t;
      if(t===last) count++;
      else { row += last+":"+count+","; last=t; count=1; }
    }
    row += last+":"+count;
    rows.push(row);
  }
  return rows;
}
function rleDecodeTiles(rows){
  const tiles=[];
  for(let y=0;y<SIZE;y++){
    tiles[y]=[];
    const parts=(rows && rows[y] ? rows[y] : "0:"+SIZE).split(",");
    let x=0;
    for(const p of parts){
      const [tv,cv]=p.split(":");
      const t=parseInt(tv,10);
      const c=parseInt(cv,10);
      for(let i=0;i<c;i++){
        tiles[y][x]=makeTile(t);
        tiles[y][x].h = (t===VOID?0:0.5);
        x++;
        if(x>=SIZE) break;
      }
      if(x>=SIZE) break;
    }
    while(x<SIZE){ tiles[y][x]=makeTile(VOID); x++; }
  }
  return tiles;
}
function exportDimensions(){
  const out={};
  for(const [id,d] of Object.entries(dimensions)){
    out[id]={
      id:d.id, name:d.name, temperament:d.temperament, biome:d.biome,
      portals:d.portals, portalBack:d.portalBack,
      myths:d.myths, history:d.history,
      tiles:rleEncodeTiles(d.tiles),
      entities:d.entities
    };
  }
  return out;
}
function importDimensions(obj){
  dimensions={};
  for(const [id,raw] of Object.entries(obj||{})){
    dimensions[id]={
      id:raw.id, name:raw.name, temperament:raw.temperament, biome:raw.biome,
      tiles:rleDecodeTiles(raw.tiles),
      entities:raw.entities||[],
      portals:raw.portals||[],
      portalBack:raw.portalBack||null,
      myths:raw.myths||null,
      history:raw.history||{visits:0, linger:0, touches:0, templeRites:0, fastTravels:0, gravity:0, lastVisitTick:0},
      discovered:true
    };
  }
}

function load(){
  const raw=localStorage.getItem(STORAGE_KEY);
  if(!raw) return false;
  try{
    const p=JSON.parse(raw);
    seedStr=p.seedStr||seedStr;
    tick=p.tick||0;
    pulse=p.pulse||0;
    Object.assign(player, p.player||player);
    travelSig=p.travelSig||travelSig;
    journal=p.journal||[];
    sigils=p.sigils||{};
    links=p.links||[];
    relics=p.relics||[];
    activeRelicId=p.activeRelicId||null;
    echoUnlocked=!!p.echoUnlocked;
    importDimensions(p.dimensions||{});
    if(echoUnlocked && !dimensions["echo"]) createDimension("echo", null, null, true);
    toast("Loaded ✅");
    return true;
  }catch(e){ return false; }
}

let last=performance.now();
let acc=0;
const step=1000/30;

function loop(t){
  const dt=t-last; last=t;
  acc+=dt;

  while(acc>=step){
    tick++;
    pulse += 0.02;

    const active=dimensions[player.dimId];
    if(active){
      active.history.linger += 0.06;
      travelSig.linger += 0.06;
      active.history.gravity = computeGravity(active);

      evolveDimension(active, true);
      relicChargeTick(active);

      updateScent(active.biome, 0.004);

      // unlock logic
      if(tick%90===0) checkEchoUnlock();
    }

    // background evolve
    const ids=Object.keys(dimensions);
    for(let i=0;i<2;i++){
      if(ids.length<2) break;
      const id=ids[(Math.random()*ids.length)|0];
      if(id!==player.dimId) evolveDimension(dimensions[id], false);
    }

    if(tick%180===0) computeLinks();

    acc-=step;
  }

  drawWorld();
  if(showMap) drawStarMap();
  if(showMyths) drawMythsPanel();
  if(showRelics) drawRelicsPanel();

  requestAnimationFrame(loop);
}
